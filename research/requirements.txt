# Research pipeline dependencies.
# Every version pinned with `==`, never `>=` (REPRODUCIBILITY.md Section 1).
# Target interpreter: Python 3.11, matching backend/pyproject.toml and Colab.
#
# Scope note: this file grows milestone by milestone, listing only what the code in
# research/src/ actually imports (CLAUDE.md rule 16). At Milestone 1 that is
# download.py plus its data-validation test. The ML stack (torch, transformers,
# scikit-learn, captum) arrives with Milestones 3-5, when the training and
# experiment scripts that import it are written.

# --- Acquisition (research/src/data/download.py) ---
requests==2.34.2
# Kaggle is the documented acquisition path for Notri-Fact (DATASET_PLAN.md
# Section 1). Requires an API token at run time; see download.py's
# _download_notri_fact docstring for the credential contract.
kaggle==2.2.4

# --- Tabular inspection (row counts in the data-validation test) ---
pandas==3.0.5
# Notri-Fact ships as a single .xlsx workbook, not CSV, so pandas needs an Excel
# engine to read it at all. Ax-to-Grind is CSV and needs none of this.
openpyxl==3.1.5

# --- Deduplication (research/src/data/dedup.py) ---
# MinHash signatures are vectorised with numpy; the pure-Python form did not finish
# in ten minutes. numpy arrives with pandas but is imported directly, so it is
# pinned explicitly rather than relied on as a transitive dependency.
numpy==2.5.2

# --- Audit (research/src/data/audit.py) ---
# blueprint Part 7 item 2 requires subword counts from the ACTUAL XLM-R tokenizer,
# not a word-count proxy — it is what the model sees and what U2 is resolved from.
# transformers is used here for its tokenizer only; torch is NOT required at this
# milestone and is deliberately not installed until Milestone 4's training runs.
transformers==5.15.0
sentencepiece==0.2.2
# Figures 1-3 (blueprint Part 31), written headless to research/results/figures/.
matplotlib==3.11.1
# Urdu rendering in figure 3. matplotlib has no Arabic shaping or bidi layout, so
# without these the required thesis figure shows isolated, unjoined letters in
# reversed order. Legible Urdu in a Urdu-language project is a stated need.
arabic-reshaper==3.0.1
python-bidi==0.6.11

# --- Classical baselines (Experiments A, B, H — Milestone 3) ---
# TF-IDF + Logistic Regression / LinearSVC, and the length-only logistic model.
scikit-learn==1.9.0
# Experiment configs are YAML (REPRODUCIBILITY.md Section 4: no run is launched
# from ad hoc CLI flags whose values aren't captured anywhere).
PyYAML==6.0.3

# --- Transformer fine-tuning (Experiments C, D — Milestone 4) ---
# The scope note at the top of this file anticipated these arriving here, and they
# are now imported by research/src/models/transformer.py.
#
# torch is pinned by VERSION only, deliberately not by build. On Linux (Colab) this
# resolves to the CUDA wheel a T4 needs; the authoring environment installed the
# same version's CPU build via `--index-url https://download.pytorch.org/whl/cpu`
# for the dry run. Same version, same behaviour, different accelerator — pinning a
# `+cpu` local would make this file uninstallable on Colab.
torch==2.13.0
# accelerate backs transformers' Trainer (device placement, fp16, dataloader prep).
accelerate==1.14.0
# huggingface_hub pushes each trained checkpoint to its staging repo (E17).
huggingface_hub==1.27.0
# transformers is already pinned above for the audit's tokenizer; the Trainer,
# AutoModelForSequenceClassification and DataCollatorWithPadding come from the
# same pin.

# --- Testing (TESTING_STRATEGY.md Section 1, "Data validation tests") ---
pytest==9.1.1
